Here is a simple struct demo.

#include <stdio.h>
#include <stdlib.h>

// 定义链表节点的结构体
typedef struct Node {
    int data;                // 节点存储的数据，这里以int类型为例
    struct Node* next;       // 指向下一个节点的指针
} Node;

// 创建新节点
Node* createNode(int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (newNode == NULL) {
        printf("内存分配失败\n");
        exit(1);
    }
    newNode->data = data;
    newNode->next = NULL;    // 新节点的next指针初始化为NULL
    return newNode;
}

int main() {
    // 创建三个独立节点
    Node* head = createNode(1);
    Node* second = createNode(2);
    Node* third = createNode(3);

    // 连接节点形成链表
    head->next = second;
    second->next = third;

    // 遍历链表并输出数据
    Node* current = head;
    while (current != NULL) {
        printf("%d ", current->data);
        current = current->next;
    }
    printf("\n");

    // 释放链表内存
    current = head;
    while (current != NULL) {
        Node* temp = current;
        current = current->next;
        free(temp);
    }

    return 0;
}
